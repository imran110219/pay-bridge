package com.paybridge.portpos;
import com.fasterxml.jackson.databind.*;import com.fasterxml.jackson.databind.node.ObjectNode;import com.paybridge.core.error.*;import com.paybridge.core.model.*;import com.paybridge.spi.*;import java.net.*;import java.net.http.*;import java.nio.charset.StandardCharsets;import java.security.*;import java.time.*;import java.util.*;
/** PortPos v2 hosted-invoice adapter (PortWallet's successor). It uses the official time-bound Bearer authentication. */
public final class PortPosPaymentGateway implements PaymentGateway {private final PortPosConfiguration c;private final HttpClient h;private final ObjectMapper j;public PortPosPaymentGateway(PortPosConfiguration c){this(c,HttpClient.newHttpClient(),new ObjectMapper());}PortPosPaymentGateway(PortPosConfiguration c,HttpClient h,ObjectMapper j){this.c=c;this.h=h;this.j=j;}public PaymentProvider provider(){return PaymentProvider.of("portpos");}public Set<GatewayCapability> capabilities(){return Set.of(GatewayCapability.CREATE_PAYMENT,GatewayCapability.GET_PAYMENT,GatewayCapability.REFUND,GatewayCapability.PARTIAL_REFUND,GatewayCapability.HOSTED_CHECKOUT,GatewayCapability.WEBHOOKS);}
public Payment createPayment(PaymentRequest r){Customer x=r.customer();if(x==null||x.phone()==null||x.phone().isBlank()||x.billingAddress()==null)throw invalid("PortPos requires customer phone and billing address");ObjectNode b=j.createObjectNode();ObjectNode o=b.putObject("order");o.put("amount",r.amount().amount().toPlainString());o.put("currency",r.amount().currency().getCurrencyCode());o.put("redirect_url",c.redirectUri().toString());o.put("reference",r.merchantReference());if(c.ipnUri()!=null)o.put("ipn_url",c.ipnUri().toString());ObjectNode p=b.putObject("product");p.put("name",text(r.description(),"Payment"));p.put("description",text(r.description(),"PayBridge payment"));ObjectNode customer=b.putObject("billing").putObject("customer");customer.put("name",x.name());customer.put("email",x.email());customer.put("phone",x.phone());BillingAddress a=x.billingAddress();ObjectNode address=customer.putObject("address");address.put("street",a.line1());address.put("city",a.city());address.put("state",a.state());address.put("zipcode",a.postalCode());address.put("country",a.country());return payment(request("POST","/payment/v2/invoice",b),r.amount());}
public Payment getPayment(PaymentReference r){return payment(request("GET","/payment/v2/invoice/"+enc(r.value()),null),null);}
public Refund refund(RefundRequest r){ObjectNode b=j.createObjectNode();ObjectNode refund=b.putObject("refund");refund.put("amount",r.amount().amount().toPlainString());refund.put("currency",r.amount().currency().getCurrencyCode());JsonNode n=request("POST","/payment/v2/invoice/refund/"+enc(r.paymentReference().value()),b);String status=n.path("data").path("order").path("status").asText();return new Refund(r.paymentReference().value(),r.paymentReference(),r.amount(),"REFUNDED".equals(status)?Refund.Status.SUCCEEDED:Refund.Status.PENDING);}
private JsonNode request(String method,String path,JsonNode body){try{long ts=Instant.now().getEpochSecond();HttpRequest.Builder b=HttpRequest.newBuilder(c.baseUri().resolve(path)).header("Authorization","Bearer "+token(ts)).header("Content-Type","application/json").header("Accept","application/json");if("GET".equals(method))b.GET();else b.POST(HttpRequest.BodyPublishers.ofString(j.writeValueAsString(body)));HttpResponse<String>r=h.send(b.build(),HttpResponse.BodyHandlers.ofString());JsonNode n=j.readTree(r.body());if(r.statusCode()>=400||!"success".equalsIgnoreCase(n.path("result").asText()))throw error(r.statusCode(),n);return n;}catch(PaymentException e){throw e;}catch(Exception e){throw new PaymentException(PaymentErrorCode.NETWORK_ERROR,provider(),null,true,"PortPos request failed");}}
private Payment payment(JsonNode n,Money requested){JsonNode d=n.path("data");JsonNode o=d.path("order");String raw=o.path("status").asText("PENDING");PaymentStatus s=switch(raw){case "ACCEPTED"->PaymentStatus.CAPTURED;case "REJECTED"->PaymentStatus.FAILED;case "CANCELLED"->PaymentStatus.CANCELLED;case "EXPIRED"->PaymentStatus.EXPIRED;case "PARTIALLY_REFUNDED"->PaymentStatus.PARTIALLY_REFUNDED;case "REFUNDED"->PaymentStatus.REFUNDED;default->PaymentStatus.PENDING;};Money amount=requested!=null?requested:Money.of(o.path("amount").asText("0"),o.path("currency").asText("BDT"));return new Payment(new PaymentReference(provider(),d.path("invoice_id").asText()),s,amount,d.path("action").path("url").asText(null),Instant.now(),Map.of("portpos_status",raw));}
private String token(long ts){return Base64.getEncoder().encodeToString((c.appKey()+":"+md5(c.secretKey()+ts)).getBytes(StandardCharsets.UTF_8));}private static String md5(String x){try{byte[]b=MessageDigest.getInstance("MD5").digest(x.getBytes(StandardCharsets.UTF_8));StringBuilder s=new StringBuilder();for(byte v:b)s.append(String.format("%02x",v));return s.toString();}catch(NoSuchAlgorithmException e){throw new IllegalStateException(e);}}private PaymentException invalid(String m){return new PaymentException(PaymentErrorCode.INVALID_REQUEST,provider(),null,false,m);}private PaymentException error(int s,JsonNode n){PaymentErrorCode code=s==401?PaymentErrorCode.AUTHENTICATION_FAILED:s==404?PaymentErrorCode.PAYMENT_NOT_FOUND:s>=500?PaymentErrorCode.PROVIDER_UNAVAILABLE:PaymentErrorCode.INVALID_REQUEST;return new PaymentException(code,provider(),null,s>=500,"PortPos returned an error");}private static String enc(String x){return URLEncoder.encode(x,StandardCharsets.UTF_8);}private static String text(String x,String d){return x==null||x.isBlank()?d:x;}}
