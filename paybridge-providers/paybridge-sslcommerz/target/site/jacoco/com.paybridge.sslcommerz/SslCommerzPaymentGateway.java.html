package com.paybridge.sslcommerz;
import com.fasterxml.jackson.databind.*;import com.paybridge.core.error.*;import com.paybridge.core.model.*;import com.paybridge.spi.*;import java.net.*;import java.net.http.*;import java.time.*;import java.util.*;
/** SSLCommerz hosted-session adapter. Applications must validate IPNs using the validation API before fulfilment. */
public final class SslCommerzPaymentGateway implements PaymentGateway {private final SslCommerzConfiguration c;private final HttpClient h;private final ObjectMapper j;public SslCommerzPaymentGateway(SslCommerzConfiguration c){this(c,HttpClient.newHttpClient(),new ObjectMapper());}SslCommerzPaymentGateway(SslCommerzConfiguration c,HttpClient h,ObjectMapper j){this.c=c;this.h=h;this.j=j;}public PaymentProvider provider(){return PaymentProvider.SSLCOMMERZ;}public Set<GatewayCapability> capabilities(){return Set.of(GatewayCapability.CREATE_PAYMENT,GatewayCapability.GET_PAYMENT,GatewayCapability.HOSTED_CHECKOUT,GatewayCapability.WEBHOOKS);}
public Payment createPayment(PaymentRequest r){if(r.customer()==null||r.customer().billingAddress()==null)throw new PaymentException(PaymentErrorCode.INVALID_REQUEST,provider(),null,false,"SSLCommerz requires customer and billing address");Map<String,String>f=new LinkedHashMap<>();f.put("store_id",c.storeId());f.put("store_passwd",c.storePassword());f.put("total_amount",r.amount().amount().toPlainString());f.put("currency",r.amount().currency().getCurrencyCode());f.put("tran_id",r.merchantReference());f.put("success_url",c.successUri().toString());f.put("fail_url",c.failureUri().toString());f.put("cancel_url",c.cancelUri().toString());f.put("ipn_url",c.ipnUri().toString());f.put("cus_name",r.customer().name());f.put("cus_email",r.customer().email());f.put("cus_add1",r.customer().billingAddress().line1());f.put("cus_city",r.customer().billingAddress().city());f.put("cus_postcode",r.customer().billingAddress().postalCode());f.put("cus_country",r.customer().billingAddress().country());f.put("shipping_method","NO");f.put("product_name",Optional.ofNullable(r.description()).orElse("Payment"));f.put("product_category","general");f.put("product_profile","general");JsonNode n=call("POST","/gwprocess/v4/api.php",f);return payment(n,r.amount(),r.merchantReference());}
public Payment getPayment(PaymentReference r){JsonNode n=call("GET","/validator/api/merchantTransIDvalidationAPI.php",Map.of("tran_id",r.value(),"store_id",c.storeId(),"store_passwd",c.storePassword(),"format","json"));if(n.has("element")&&n.get("element").isArray()&&!n.get("element").isEmpty())n=n.get("element").get(0);return payment(n,null,r.value());}
private JsonNode call(String method,String path,Map<String,String>f){try{URI u=c.baseUri().resolve(path);HttpRequest req;if("GET".equals(method))req=HttpRequest.newBuilder(URI.create(u+"?"+form(f))).GET().build();else req=HttpRequest.newBuilder(u).header("Content-Type","application/x-www-form-urlencoded").POST(HttpRequest.BodyPublishers.ofString(form(f))).build();HttpResponse<String>res=h.send(req,HttpResponse.BodyHandlers.ofString());JsonNode n=j.readTree(res.body());if(res.statusCode()>=400||"FAILED".equals(n.path("APIConnect").asText()))throw new PaymentException(res.statusCode()==401?PaymentErrorCode.AUTHENTICATION_FAILED:PaymentErrorCode.PROVIDER_UNAVAILABLE,provider(),null,res.statusCode()>=500,"SSLCommerz request failed");return n;}catch(PaymentException e){throw e;}catch(Exception e){throw new PaymentException(PaymentErrorCode.NETWORK_ERROR,provider(),null,true,"SSLCommerz request failed");}}
private Payment payment(JsonNode n,Money requested,String reference){String raw=n.path("status").asText();PaymentStatus s=switch(raw){case "VALID","VALIDATED"->PaymentStatus.CAPTURED;case "FAILED"->PaymentStatus.FAILED;case "CANCELLED"->PaymentStatus.CANCELLED;case "EXPIRED"->PaymentStatus.EXPIRED;case "UNATTEMPTED"->PaymentStatus.CREATED;default->PaymentStatus.PENDING;};Money a=requested!=null?requested:Money.of(n.path("amount").asText("0"),n.path("currency").asText(n.path("currency_type").asText("BDT")));Map<String,String>m=new HashMap<>();if(n.hasNonNull("sessionkey"))m.put("session_key",n.get("sessionkey").asText());if(n.hasNonNull("bank_tran_id"))m.put("bank_transaction_id",n.get("bank_tran_id").asText());m.put("sslcommerz_status",raw);return new Payment(new PaymentReference(provider(),reference),s,a,n.path("GatewayPageURL").asText(null),Instant.now(),m);}private static String form(Map<String,String>f){return f.entrySet().stream().map(e->URLEncoder.encode(e.getKey(),java.nio.charset.StandardCharsets.UTF_8)+"="+URLEncoder.encode(e.getValue(),java.nio.charset.StandardCharsets.UTF_8)).collect(java.util.stream.Collectors.joining("&"));}}
